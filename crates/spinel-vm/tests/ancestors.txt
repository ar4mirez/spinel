# Ancestor ordering, as measured from CRuby.
#
# Every `expect` line in this file was produced by running the case on the real
# Ruby named in CI and recording the answer. `scripts/ancestors-oracle.rb` is
# what produces and re-checks them; `ancestors.rs` is what makes spinel-vm agree.
#
# So this file is the definition of done for #8, and it is a measurement rather
# than a transcription: `include`/`prepend` ordering follows rules CRuby
# documents nowhere, and several of them are surprising enough that reading
# `class.c` and believing the result would have been the wrong move.
#
# Verbs: `module X`, `class X [< Y]`, `include X A B`, `prepend X A`,
# `singleton X SX`, `expect X : ...`. An `expect` keeps only the entities its
# case declared, plus any built-in it names itself.

# ---------------------------------------------------------------- the basics

case one_include
  module M
  class C
  include C M
  expect C : C M
end

case a_later_include_sits_closer
  module A
  module B
  class C
  include C A
  include C B
  expect C : C B A
end

case one_include_argument_list_is_reversed
  module A
  module B
  module D
  class C
  include C A B D
  expect C : C A B D
end

case an_included_module_brings_its_own_chain
  module A
  module B
  module D
  class C
  include B A
  include D B
  include C D
  expect C : C D B A
end

case a_repeated_include_changes_nothing
  module A
  module B
  class C
  include C A
  include C B
  include C A
  expect C : C B A
end

# ruby/spec's `Module#ancestors` opener, entity for entity.
case a_superclass_chain_follows_the_class
  module Basic
  module Super
  class Parent
  class Child < Parent
  include Super Basic
  include Child Super
  expect Super : Super Basic
  expect Child : Child Super Basic Parent Object Kernel BasicObject
end
